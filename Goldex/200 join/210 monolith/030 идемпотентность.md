Идемпотентность в **Node.js cluster** почти всегда ломается по одной причине: **у воркеров нет общей памяти**, а запрос/задача может прилететь **повторно** (ретраи клиента, таймауты, повторная доставка из очереди, повторная отправка webhook’а) и попасть **в другой процесс**.

Поэтому “сделать функцию идемпотентной” в кластере = **добавить общий (shared) источник правды** + **атомарную дедупликацию**.

Ниже — рабочие подходы (от самых надёжных).

---

## 1) Самый надёжный способ: идемпотентность через БД (unique key + транзакция)

Если у тебя есть PostgreSQL/MySQL — это обычно лучший вариант, потому что:

- БД умеет **атомарность** и **блокировки**
    
- не нужно городить distributed lock
    
- проще переживать рестарты воркеров
    

### Идея

1. Клиент присылает `Idempotency-Key` (или ты сам вычисляешь ключ).
    
2. Ты сохраняешь этот ключ в таблицу **idempotency_keys**.
    
3. Если ключ уже был — возвращаешь **сохранённый результат**.
    
4. Если ключ новый — выполняешь работу и сохраняешь результат.
    

### Важно

Чтобы ключ **нельзя было “переиспользовать” с другим телом запроса**, сохраняй ещё `request_hash` и сравнивай.

### DDL пример (PostgreSQL)

```sql
CREATE TABLE idempotency_keys (
  key TEXT PRIMARY KEY,
  request_hash TEXT NOT NULL,
  response_status INT,
  response_body JSONB,
  created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
  completed_at TIMESTAMPTZ
);

-- полезно чистить старые ключи по TTL:
CREATE INDEX ON idempotency_keys (created_at);
```

### Node.js (pg) — шаблон

```js
import crypto from "crypto";
import { Pool } from "pg";

const pool = new Pool({ connectionString: process.env.DATABASE_URL });

function hashRequest(body) {
  return crypto.createHash("sha256").update(JSON.stringify(body)).digest("hex");
}

export async function runIdempotent({ key, requestHash }, handler) {
  const client = await pool.connect();
  try {
    await client.query("BEGIN");

    // 1) создаём запись (если новая)
    await client.query(
      `INSERT INTO idempotency_keys(key, request_hash)
       VALUES ($1, $2)
       ON CONFLICT (key) DO NOTHING`,
      [key, requestHash]
    );

    // 2) лочим строку по ключу, чтобы параллельные повторы ждали
    const { rows } = await client.query(
      `SELECT request_hash, response_status, response_body
       FROM idempotency_keys
       WHERE key = $1
       FOR UPDATE`,
      [key]
    );

    const row = rows[0];
    if (!row) throw new Error("Idempotency row missing");

    // защита от повторного использования ключа с другим запросом
    if (row.request_hash !== requestHash) {
      await client.query("ROLLBACK");
      const err = new Error("Idempotency-Key reused with different request body");
      err.code = "IDEMPOTENCY_MISMATCH";
      throw err;
    }

    // 3) если ответ уже сохранён — возвращаем его
    if (row.response_status != null) {
      await client.query("COMMIT");
      return { status: row.response_status, body: row.response_body };
    }

    // 4) иначе выполняем работу в ЭТОЙ ЖЕ транзакции
    const result = await handler(client);

    await client.query(
      `UPDATE idempotency_keys
       SET response_status = $2,
           response_body = $3,
           completed_at = now()
       WHERE key = $1`,
      [key, result.status, result.body]
    );

    await client.query("COMMIT");
    return result;
  } catch (e) {
    try { await client.query("ROLLBACK"); } catch {}
    throw e;
  } finally {
    client.release();
  }
}
```

### Использование в HTTP (Express/Fastify — идея одинаковая)

```js
app.post("/orders", async (req, res, next) => {
  try {
    const idem = req.get("Idempotency-Key");
    if (!idem) return res.status(400).send("Missing Idempotency-Key");

    const key = `${req.user.id}:${idem}`;           // обязательно привяжи к юзеру/тенанту
    const requestHash = hashRequest(req.body);

    const result = await runIdempotent({ key, requestHash }, async (tx) => {
      // любые запросы в БД через tx
      const { rows } = await tx.query(
        `INSERT INTO orders(user_id, payload)
         VALUES ($1, $2)
         RETURNING id`,
        [req.user.id, req.body]
      );

      return { status: 201, body: { orderId: rows[0].id } };
    });

    res.status(result.status).json(result.body);
  } catch (err) {
    if (err.code === "IDEMPOTENCY_MISMATCH") {
      return res.status(409).json({ error: err.message });
    }
    next(err);
  }
});
```

## 2) Redis (SET NX + хранение результата) — быстро, но аккуратно с “крашем между шагами”

Redis-подход часто используют, когда нет удобной БД-логики или нужна высокая скорость.

### Базовая схема

- `SET idem:lock:<key> token NX PX ttl` — захват “в процессе”
    
- по завершении — `SET idem:result:<key> <json> PX ttl`
    
- повтор — если есть result → верни его
    

### Главный риск

Если ты сделал **внешний сайд-эффект** (например списал деньги), а воркер **упал до сохранения результата** в Redis — следующий повтор может сделать сайд-эффект второй раз.

Поэтому Redis-идемпотентность **идеальна для:**

- задач, где сайд-эффект тоже идемпотентен
    
- или когда внешний сервис поддерживает свой `idempotency_key` (и ты его туда прокидываешь)
    

### Пример (ioredis)

```js
import crypto from "crypto";
import Redis from "ioredis";

const redis = new Redis(process.env.REDIS_URL);

const releaseLua = `
if redis.call("get", KEYS[1]) == ARGV[1]
then return redis.call("del", KEYS[1])
else return 0 end
`;

export async function withIdempotency(key, ttlMs, fn) {
  const lockKey = `idem:lock:${key}`;
  const resKey  = `idem:res:${key}`;

  // 1) быстрый путь
  const cached = await redis.get(resKey);
  if (cached) return JSON.parse(cached);

  // 2) захват "в процессе"
  const token = crypto.randomUUID();
  const acquired = await redis.set(lockKey, token, "NX", "PX", ttlMs);
  if (!acquired) {
    // другой воркер делает то же самое
    // вариант: подождать результат коротким polling’ом
    for (let i = 0; i < 20; i++) {
      await new Promise(r => setTimeout(r, 100));
      const now = await redis.get(resKey);
      if (now) return JSON.parse(now);
    }
    // или вернуть 409/202
    throw new Error("Already in progress");
  }

  try {
    const result = await fn();
    await redis.set(resKey, JSON.stringify(result), "PX", ttlMs);
    return result;
  } finally {
    await redis.eval(releaseLua, 1, lockKey, token);
  }
}
```

## 3) Делай операцию “естественно идемпотентной” (если можешь)

Иногда можно обойтись вообще без таблицы ключей, если бизнес-операция сама по себе идемпотентна:

- **PUT вместо POST**: “установить состояние” вместо “создать ещё раз”
    
- **upsert** по естественному ключу: `order_id`, `payment_id`, `external_event_id`
    
- запись с **детерминированным ID**: например `orderId = hash(userId + cartId)`
    

Пример: `INSERT ... ON CONFLICT DO NOTHING` + `SELECT` уже созданного объекта.

---

## 4) Если есть внешние эффекты (платежи/почта/интеграции): прокидывай idempotency key “вниз” + outbox

Для надёжности в проде обычно делают так:

1. В БД в транзакции:
    

- создаёшь заказ/сущность
    
- пишешь событие в `outbox` (уникальный event_id)
    

2. Отдельный воркер читает outbox и отправляет письмо/вебхук/платёж:
    

- с **dedupe** по `event_id`
    
- и/или использует **idempotency key** внешнего сервиса
    

Так ты избегешь “сделал сайд-эффект, но не успел зафиксировать результат”.

---

## 5) Почему не “просто хранить в памяти воркера” / sticky sessions?

В кластере это не работает, потому что:

- повторный запрос может уйти в другой воркер
    
- воркер может перезапуститься
    
- при горизонтальном масштабировании (несколько машин) вообще теряется смысл
    

Идемпотентность должна жить в **общем хранилище**.

## Практический чек-лист (коротко)

**Идемпотенси-ключ**

- генерирует клиент и шлёт в `Idempotency-Key` (часто лучший вариант)
    
- либо ты генерируешь из бизнес-идентификатора (orderId/paymentId)
    

**Ключ обязательно связывай**

- с userId/tenantId (`userId:key`)
    
- с endpoint’ом (чтобы один ключ не работал на разные операции)
    

**Защита от злоупотреблений**

- храни `request_hash` и возвращай `409 Conflict`, если ключ тот же, а тело другое
    

**TTL**

- храни ключи достаточно долго (например 24 часа для платежей, меньше для простых операций)
    
- чисти кроном/задачей
    

**Параллельные повторы**

- в БД: `SELECT ... FOR UPDATE`
    
- в Redis: “in progress” lock + ожидание результата
